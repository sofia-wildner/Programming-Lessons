    <!-- O "Cérebro" do Carrossel -->
    <script>
        // 1. Mapeamento dos elementos do DOM
        const trilha = document.getElementById('trilha');
        const btnVoltar = document.getElementById('btn-voltar');
        const btnAvancar = document.getElementById('btn-avancar');
        const slides = document.querySelectorAll('.slide');
        
        // 2. Estado inicial
        let indiceAtual = 0;
        const totalSlides = slides.length;

        // 3. Função central que move a trilha
        function atualizarCarrossel() {
            // Se o índice for 1, move -100%. Se for 2, move -200%...
            const deslocamento = indiceAtual * -100;
            trilha.style.transform = `translateX(${deslocamento}%)`;
        }

        // 4. Lógica para avançar
        btnAvancar.addEventListener('click', () => {
            if (indiceAtual === totalSlides - 1) {
                // Se estiver no último, volta pro primeiro
                indiceAtual = 0;
            } else {
                // Senão, apenas avança um
                indiceAtual++;
            }
            atualizarCarrossel();
        });

        // 5. Lógica para voltar
        btnVoltar.addEventListener('click', () => {
            if (indiceAtual === 0) {
                // Se estiver no primeiro, vai pro último
                indiceAtual = totalSlides - 1;
            } else {
                // Senão, apenas volta um
                indiceAtual--;
            }
            atualizarCarrossel();
        });
    </script>
