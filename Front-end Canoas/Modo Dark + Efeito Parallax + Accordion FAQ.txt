    // Único trecho de JavaScript de toda a aplicação!
    const botaoTema = document.getElementById('btn-tema');
    
    botaoTema.addEventListener('click', () => {
      // O toggle adiciona a classe se ela não existir, e remove se existir
      document.body.classList.toggle('dark-theme');
      
      // Muda o texto e o ícone do botão dependendo do tema atual
      if (document.body.classList.contains('dark-theme')) {
        botaoTema.textContent = 'Modo Claro ☀️';
      } else {
        botaoTema.textContent = 'Modo Escuro 🌙';
      }
    });
